cmake_minimum_required(VERSION 3.20)
# Active Field Analyzer project. Frozen M2 root is archive/m2-ui/CMakeLists.txt.
option(FIELD_BUILD_DEVICE "Build a DISPLAY SRAM candidate instead of host previews" OFF)
option(FIELD_STANDARD_STARTUP_APPROVED "Explicit offline-build exception only; no physical startup/deployment approval" OFF)
set(WILIBSP_PATH "${CMAKE_CURRENT_LIST_DIR}/wilibsp" CACHE PATH "Unmodified pinned WiliBSP sources")
if(FIELD_BUILD_DEVICE)
    if(NOT FIELD_STANDARD_STARTUP_APPROVED)
        message(FATAL_ERROR "BSP startup changes VREF/power/internal GPIO. Host UI is supported; obtain explicit startup approval before a device candidate.")
    endif()
    set(PICO_BOARD freewili2 CACHE STRING "Board type")
    list(APPEND PICO_BOARD_HEADER_DIRS "${WILIBSP_PATH}/bsp/boards")
    set(PICO_PIO_USB_PATH "${WILIBSP_PATH}/bsp/third_party/Pico-PIO-USB")
    set(PICO_SDK_FETCH_FROM_GIT OFF CACHE BOOL "Explicit SDK sources only")
    include("${WILIBSP_PATH}/pico_sdk_import.cmake")
    project(WiliPirate C CXX ASM)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    set(FW2_AGENTIO OFF CACHE BOOL "No remote input harness" FORCE)
    pico_sdk_init()
    if(NOT PICO_SDK_VERSION_STRING STREQUAL "2.3.0")
        message(FATAL_ERROR "Pinned Field Analyzer requires Pico SDK 2.3.0")
    endif()
else()
    project(WiliPirateFieldAnalyzerHost C)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(field_ui STATIC native/field_analyzer/model.c native/field_analyzer/ui.c native/field_analyzer/can_model.c)
target_include_directories(field_ui PUBLIC native/field_analyzer)
target_compile_options(field_ui PRIVATE -Wall -Wextra -Werror)
if(FIELD_BUILD_DEVICE)
    add_subdirectory("${WILIBSP_PATH}/bsp" "${CMAKE_BINARY_DIR}/bsp")
    add_subdirectory(native/field_analyzer)
else()
    enable_testing()
    add_executable(field_navigation_test native/field_analyzer/tests/navigation_test.c)
    target_link_libraries(field_navigation_test PRIVATE field_ui)
    add_test(NAME field_navigation COMMAND field_navigation_test)
    add_executable(field_can_test native/field_analyzer/tests/can_test.c)
    target_link_libraries(field_can_test PRIVATE field_ui)
    add_test(NAME field_can COMMAND field_can_test)
    add_executable(field_render_test native/field_analyzer/tests/render_test.c "${WILIBSP_PATH}/bsp/display/font5x7.c")
    target_include_directories(field_render_test PRIVATE "${WILIBSP_PATH}/bsp")
    target_link_libraries(field_render_test PRIVATE field_ui)
    add_test(NAME field_render COMMAND field_render_test)
endif()
