cmake_minimum_required(VERSION 3.24)
project(WiliPirateCM0 LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "CM0 binary requires Linux; use an ARM64 Linux toolchain/sysroot on the host")
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  message(FATAL_ERROR "Deployment target must be ARM64 Linux, not a desktop executable")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(WILICM0_BUILD_DRIVER OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../vendor/wilicm0bsp" "${CMAKE_BINARY_DIR}/wilicm0bsp")
add_executable(WiliPirate main.cpp)
target_link_libraries(WiliPirate PRIVATE wilicm0::wilicm0)
target_compile_definitions(WiliPirate PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(WiliPirate PRIVATE -Wall -Wextra -Werror)
# Host staging only: never install to a device implicitly.
install(TARGETS WiliPirate RUNTIME DESTINATION wilipirate)
install(PROGRAMS run.sh DESTINATION wilipirate)
install(FILES README.md DESTINATION wilipirate)
install(FILES ../../vendor/wilicm0bsp/LICENSE DESTINATION wilipirate/licenses RENAME WiliCM0BSP-LICENSE)
install(FILES ../../vendor/wilicm0bsp/libs/onewili/LICENSE DESTINATION wilipirate/licenses RENAME OneWili-LICENSE)
