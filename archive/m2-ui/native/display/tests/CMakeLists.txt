cmake_minimum_required(VERSION 3.20)
project(WiliPirateDisplayHostTests C)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(WP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(navigation_test navigation_test.c)
target_include_directories(navigation_test PRIVATE ..)
target_compile_options(navigation_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME navigation COMMAND navigation_test)
add_executable(render_test render_test.c ../ui.c
    "${WP_ROOT}/wilibsp/bsp/display/font5x7.c")
target_include_directories(render_test PRIVATE .. "${WP_ROOT}/wilibsp/bsp")
target_compile_options(render_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME rendering COMMAND render_test)
